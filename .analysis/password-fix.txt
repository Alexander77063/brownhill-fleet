// /tmp/password-patch.py — patch password.ts pbkdf2 to encode as UTF-8
import sys

src = '/opt/brownhill/repo/src/lib/auth/password.ts'
with open(src) as f:
    s = f.read()

old = '''async function pbkdf2(password: string, salt: Uint8Array<ArrayBuffer>, iterations: number, keylenBytes: number): Promise<Uint8Array<ArrayBuffer>> {
  const pwBuf = new ArrayBuffer(password.length * 4);
  const pwBytes = new Uint8Array(pwBuf);
  for (let i = 0; i < password.length; i++) pwBytes[i] = password.charCodeAt(i);
  const km = await crypto.subtle.importKey(
    'raw',
    pwBytes,
    'PBKDF2',
    false,
    ['deriveBits']
  );
  const bits = await crypto.subtle.deriveBits(
    { name: 'PBKDF2', salt, iterations, hash: PARAMS.hash },
    km,
    keylenBytes * 8,
  );
  // bits is ArrayBuffer; wrap in Uint8Array<ArrayBuffer>
  return new Uint8Array(bits);
}'''

new = '''async function pbkdf2(password: string, salt: Uint8Array<ArrayBuffer>, iterations: number, keylenBytes: number): Promise<Uint8Array<ArrayBuffer>> {
  // Encode the password as UTF-8 — same default Node's `crypto.pbkdf2Sync`
  // applies when given a string. The previous version allocated
  // `password.length * 4` bytes and only zero-filled them out to UTF-16 LE
  // low-byte; the trailing zeros were silently included as key material,
  // so a hash produced by the in-browser implementation could never
  // verify against a hash produced by Node's implementation and the
  // customer's pre-deploy password hashes were effectively unreadable.
  const pwBytes = new TextEncoder().encode(password);
  const km = await crypto.subtle.importKey(
    'raw',
    pwBytes as BufferSource,
    'PBKDF2',
    false,
    ['deriveBits']
  );
  const bits = await crypto.subtle.deriveBits(
    { name: 'PBKDF2', salt, iterations, hash: PARAMS.hash },
    km,
    keylenBytes * 8,
  );
  // bits is ArrayBuffer; wrap in Uint8Array<ArrayBuffer>
  return new Uint8Array(bits);
}'''

if old not in s:
    print('OLD NOT FOUND — file may already be patched or changed', file=sys.stderr)
    sys.exit(2)

if new in s:
    print('ALREADY PATCHED')
    sys.exit(0)

s = s.replace(old, new, 1)
with open(src, 'w') as f:
    f.write(s)
print('OK patched')
